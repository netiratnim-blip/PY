1. วัตถุประสงค์: เก็บรวบรวมและใช้ข้อมูลชื่อ-นามสกุล เลขบัตรประชาชน เบอร์โทร อีเมล บัญชีธนาคาร และสำเนาเอกสาร เพื่อประเมินสินเชื่อ ยืนยันตัวตน และติดต่อกลับ

2. การเปิดเผยข้อมูล: ข้อมูลของท่านจะถูกเก็บรักษาเป็นความลับตามมาตรฐานความปลอดภัย และไม่นำไปเผยแพร่แก่บุคคลภายนอกโดยมิชอบ

3. สิทธิของเจ้าของข้อมูล: ท่านมีสิทธิขอเข้าถึง แก้ไข ลบ หรือถอนความยินยอมได้ตลอดเวลาโดยติดต่อเจ้าหน้าที่

1. เปิด Google Sheet > เมนู ส่วนขยาย (Extensions) > Apps Script
2. วางโค้ดนี้ แก้ไขอีเมลใน ADMIN_EMAIL แล้วกด Deploy > Web app (Anyone)

const ADMIN_EMAIL = "YOUR_EMAIL@gmail.com";

function doPost(e) {
  const data = JSON.parse(e.postData.contents);
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("Loans") || ss.insertSheet("Loans");
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["วันที่", "รหัสกู้", "สถานะ", "ชื่อ-สกุล", "เลขบัตร", "โทร", "อีเมล", "อาชีพ", "ธนาคาร", "รูปแบบ", "วงเงิน", "ค่างวด", "ลิงก์บัตร ปชช.", "ลิงก์เอกสารรายได้"]);
  }
  const folder = DriveApp.getFoldersByName("LoanDocs").hasNext() ? DriveApp.getFoldersByName("LoanDocs").next() : DriveApp.createFolder("LoanDocs");
  let idUrl = "-", incUrl = "-", files = [];
  if (data.idCardBase64) {
    const b1 = Utilities.newBlob(Utilities.base64Decode(data.idCardBase64.split(",")[1]), "image/jpeg", data.loanId + "_IDCard.jpg");
    const f1 = folder.createFile(b1); f1.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    idUrl = f1.getUrl(); files.push(b1);
  }
  if (data.incomeBase64) {
    const b2 = Utilities.newBlob(Utilities.base64Decode(data.incomeBase64.split(",")[1]), "image/jpeg", data.loanId + "_Income.jpg");
    const f2 = folder.createFile(b2); f2.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    incUrl = f2.getUrl(); files.push(b2);
  }
  sheet.appendRow([data.timestamp, data.loanId, data.status, data.fullName, "'" + data.nationalId, "'" + data.phone, data.email, data.occupation, data.bankInfo, data.repayType, data.amount, data.installmentText, idUrl, incUrl]);
  MailApp.sendEmail({
    to: data.adminEmail || ADMIN_EMAIL,
    subject: "[คำขอสินเชื่อใหม่] " + data.loanId + " - " + data.fullName,
    body: "รหัสกู้: " + data.loanId + "\nผู้กู้: " + data.fullName + "\nโทร: " + data.phone + "\nวงเงิน: " + data.amount + " บาท (" + data.repayType + ")\nค่างวด: " + data.installmentText + "\nบัญชี: " + data.bankInfo + "\nสำเนาบัตร ปชช.: " + idUrl,
    attachments: files
  });
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
