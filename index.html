วิธีที่ 1: เชื่อมต่อกับ Google Form โดยตรง (ไม่ต้องเขียนโค้ด)

                    
    
                        
  1. สร้าง Google Form ที่มีคำถาม: ชื่อผู้กู้, เลขบัตรประชาชน, เบอร์โทร, จำนวนเงินกู้
    2. 
                        
  2. กดปุ่ม จุด 3 จุดมุมขวาบน เลือก "รับลิงก์ที่กรอกข้อมูลไว้ล่วงหน้า" (Get pre-filled link)
    3. 
                        
  3. กรอกข้อมูลทดสอบแล้วกดรับลิงก์ นำลิงก์มาดูรหัส entry.xxxxxxx ของแต่ละช่อง
    4. 
                        
  4. นำ URL ของฟอร์มเปลี่ยนท้ายจาก /viewform เป็น /formResponse มาวางในช่องตั้งค่าข้อ 4
    5. 
                    

                


                

                    
วิธีที่ 2: ส่งเข้า Google Sheets ผ่าน Google Apps Script (แนะนำ)

                    
เปิด Google Sheets > เมนู ส่วนขยาย (Extensions) > Apps Script วางโค้ดนี้แล้วกด Deploy เป็น Web App (สิทธิ์ Everyone):

                    
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(), data.contractNumber, data.borrowerName,
    data.borrowerIdCard, data.borrowerPhone, data.loanAmount,
    data.interestRate, data.dueDate, data.lenderName
  ]);
  return ContentService.createTextOutput(JSON.stringify({"status":"success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
