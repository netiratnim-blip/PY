หนังสือขอความยินยอมและแจ้งนโยบายความเป็นส่วนตัว

ตามพระราชบัญญัติคุ้มครองข้อมูลส่วนบุคคล พ.ศ. 2562 (PDPA)

บังคับอ่านและยอมรับ

ผู้ให้บริการสินเชื่อ ("ผู้ควบคุมข้อมูลส่วนบุคคล") ให้ความสำคัญอย่างยิ่งต่อการคุ้มครองข้อมูลส่วนบุคคลของท่าน เพื่อให้เป็นไปตามพระราชบัญญัติคุ้มครองข้อมูลส่วนบุคคล พ.ศ. 2562 (PDPA) กรุณาอ่านรายละเอียดต่อไปนี้ก่อนให้ความยินยอม:

1. วัตถุประสงค์ในการเก็บรวบรวม ใช้ และเปิดเผยข้อมูล

  • เพื่อการพิจารณาอนุมัติวงเงินสินเชื่อ ประเมินความสามารถในการชำระหนี้ และตรวจสอบคุณสมบัติผู้กู้
  • เพื่อการยืนยันตัวตน (KYC) ป้องกันการทุจริตและการแอบอ้างสิทธิ์
  • เพื่อการติดต่อสื่อสาร แจ้งผลการพิจารณา นัดหมายทำสัญญากู้ยืมเงิน และติดตามทวงถามหนี้ตามกฎหมาย
  • เพื่อการจัดทำสัญญากู้ยืมเงินและปฏิบัติตามข้อกำหนดทางกฎหมายที่เกี่ยวข้อง

2. ประเภทข้อมูลส่วนบุคคลที่จัดเก็บ

ชื่อ-นามสกุล, เลขประจำตัวประชาชน, วันเดือนปีเกิด, ที่อยู่ปัจจุบัน, หมายเลขโทรศัพท์, อีเมล, อาชีพและรายได้, ข้อมูลบุคคลอ้างอิงฉุกเฉิน, ภาพถ่ายสำเนาบัตรประจำตัวประชาชน และเอกสารแสดงรายได้ (หากมีการแนบ)

3. ระยะเวลาการเก็บรักษาข้อมูล

เราจะเก็บรักษาข้อมูลส่วนบุคคลของท่านไว้ตลอดระยะเวลาที่เป็นลูกค้าหรือมีนิติสัมพันธ์ตามสัญญา และจะเก็บต่อเนื่องไปอีกไม่เกิน 10 ปี นับแต่วันที่สิ้นสุดสัญญาตามอายุความทางกฎหมาย (หรือหากไม่ได้รับการอนุมัติ จะทำลายข้อมูลภายใน 90 วัน เว้นแต่ท่านจะขอให้ลบก่อนกำหนด)

4. สิทธิของเจ้าของข้อมูลส่วนบุคคล

ท่านมีสิทธิในการขอเข้าถึง ขอรับสำเนา ขอแก้ไขข้อมูลให้ถูกต้อง ขอให้ลบหรือทำลายข้อมูล ขอระงับการใช้ข้อมูล รวมถึงมีสิทธิถอนความยินยอมเมื่อใดก็ได้ โดยการถอนความยินยอมจะไม่ส่งผลกระทบต่อการประมวลผลข้อมูลที่ได้ดำเนินการไปแล้วโดยชอบด้วยกฎหมาย

ช่องทางติดต่อผู้ควบคุมข้อมูลส่วนบุคคล (DPO): หากท่านต้องการใช้สิทธิหรือสอบถามเพิ่มเติม สามารถติดต่อได้ที่อีเมลผู้ดูแลระบบที่ระบุไว้ในหน้าเว็บไซต์ หรือแจ้งผ่านเจ้าหน้าที่ผู้ติดต่อกลับ

const ADMIN_EMAIL = "YOUR_EMAIL_HERE@gmail.com"; // <-- ใส่อีเมลของคุณที่นี่
const DRIVE_FOLDER_NAME = "Loan_Documents_Uploads";

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName("LoanApplications");
    if (!sheet) {
      sheet = ss.insertSheet("LoanApplications");
      sheet.appendRow([
        "วันที่ยื่นคำขอ", "รหัสอ้างอิง (Loan ID)", "สถานะ", "หมายเหตุเจ้าหน้าที่",
        "ชื่อ-นามสกุล", "เลขบัตรประชาชน", "วันเกิด", "เบอร์โทร", "อีเมล", "LINE ID",
        "อาชีพ", "รายได้ต่อเดือน", "สถานที่ทำงาน", "ที่อยู่ปัจจุบัน",
        "บุคคลอ้างอิง", "รูปแบบการผ่อน", "วงเงินกู้", "ระยะเวลา", "สรุปยอดผ่อน",
        "วัตถุประสงค์", "ลิงก์สำเนาบัตรประชาชน", "ลิงก์เอกสารรายได้", "สถานะ PDPA"
      ]);
    }

    // กรณีเจ้าหน้าที่อัปเดตสถานะจากหน้าเว็บ
    if (data.action === "updateStatus") {
      const rows = sheet.getDataRange().getValues();
      for (let i = 1; i < rows.length; i++) {
        if (rows[i][1] === data.loanId) {
          sheet.getRange(i + 1, 3).setValue(data.status);
          sheet.getRange(i + 1, 4).setValue(data.note || "");
          return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON);
        }
      }
    }

    // สร้างโฟลเดอร์เก็บเอกสารใน Google Drive
    const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
    const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME);

    let idCardUrl = "ไม่มีไฟล์";
    let incomeDocUrl = "ไม่ได้แนบ";
    let attachments = [];

    // บันทึกไฟล์สำเนาบัตรประชาชนลง Drive และเตรียมแนบอีเมล
    if (data.idCardFile && data.idCardFile.base64) {
      const blob1 = Utilities.newBlob(Utilities.base64Decode(data.idCardFile.base64), data.idCardFile.mimeType, data.loanId + "_IDCard_" + data.idCardFile.name);
      const file1 = folder.createFile(blob1);
      file1.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      idCardUrl = file1.getUrl();
      attachments.push(blob1);
    }

    // บันทึกไฟล์เอกสารรายได้ลง Drive และเตรียมแนบอีเมล
    if (data.incomeDocFile && data.incomeDocFile.base64) {
      const blob2 = Utilities.newBlob(Utilities.base64Decode(data.incomeDocFile.base64), data.incomeDocFile.mimeType, data.loanId + "_Income_" + data.incomeDocFile.name);
      const file2 = folder.createFile(blob2);
      file2.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      incomeDocUrl = file2.getUrl();
      attachments.push(blob2);
    }

    // บันทึกข้อมูลลง Google Sheet
    sheet.appendRow([
      data.timestamp, data.loanId, data.status || "รอตรวจสอบ", "",
      data.fullName, "'" + data.nationalId, data.birthDate, "'" + data.phone, data.email, data.lineId,
      data.occupation, data.monthlyIncome, data.workplace, data.currentAddress,
      data.referencePerson, data.loanType, data.loanAmount, data.loanTerm, data.installmentSummary,
      data.loanPurpose, idCardUrl, incomeDocUrl, data.pdpaStatus
    ]);

    // ส่งอีเมลแจ้งเตือนแอดมินอัตโนมัติ พร้อมแนบไฟล์รูปภาพ/เอกสาร
    const emailSubject = "[คำขอสินเชื่อใหม่] " + data.loanId + " - คุณ" + data.fullName + " (" + data.loanAmount + " บาท)";
    const emailBody = 
      "มีรายการยื่นคำขอสินเชื่อใหม่เข้ามาในระบบ:\n\n" +
      "รหัสอ้างอิง: " + data.loanId + "\n" +
      "ชื่อ-นามสกุล: " + data.fullName + "\n" +
      "เลขบัตรประชาชน: " + data.nationalId + "\n" +
      "เบอร์โทรศัพท์: " + data.phone + "\n" +
      "อีเมล: " + data.email + "\n" +
      "LINE ID: " + (data.lineId || "-") + "\n\n" +
      "--- รายละเอียดการกู้ ---\n" +
      "รูปแบบ: " + data.loanType + "\n" +
      "วงเงินที่ขอกู้: " + data.loanAmount + " บาท\n" +
      "สรุปยอดผ่อน: " + data.installmentSummary + "\n" +
      "อาชีพ/รายได้: " + data.occupation + " (" + data.monthlyIncome + " บาท/เดือน)\n" +
      "ที่อยู่: " + data.currentAddress + "\n\n" +
      "ลิงก์สำเนาบัตรประชาชน (Google Drive): " + idCardUrl + "\n" +
      "ลิงก์เอกสารรายได้ (Google Drive): " + incomeDocUrl + "\n\n" +
      "(ไฟล์เอกสารต้นฉบับถูกแนบมากับอีเมลฉบับนี้แล้ว)";

    MailApp.sendEmail({
      to: ADMIN_EMAIL,
      subject: emailSubject,
      body: emailBody,
      attachments: attachments
    });

    return ContentService.createTextOutput(JSON.stringify({ success: true, idCardUrl: idCardUrl, incomeDocUrl: incomeDocUrl })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("LoanApplications");
  if (!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  
  const rows = sheet.getDataRange().getValues();
  const list = [];
  for (let i = 1; i < rows.length; i++) {
    list.push({
      timestamp: rows[i][0],
      loanId: rows[i][1],
      status: rows[i][2],
      staffNote: rows[i][3],
      fullName: rows[i][4],
      nationalId: String(rows[i][5]).replace(/^'/, ""),
      birthDate: rows[i][6],
      phone: String(rows[i][7]).replace(/^'/, ""),
      email: rows[i][8],
      lineId: rows[i][9],
      occupation: rows[i][10],
      monthlyIncome: rows[i][11],
      workplace: rows[i][12],
      currentAddress: rows[i][13],
      referencePerson: rows[i][14],
      loanType: rows[i][15],
      loanAmount: rows[i][16],
      loanTerm: rows[i][17],
      installmentSummary: rows[i][18],
      loanPurpose: rows[i][19],
      idCardUrl: rows[i][20],
      incomeDocUrl: rows[i][21]
    });
  }
  return ContentService.createTextOutput(JSON.stringify(list.reverse())).setMimeType(ContentService.MimeType.JSON);
}
